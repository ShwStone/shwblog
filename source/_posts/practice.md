$$
\begin{aligned}
&\lim_{n \to \infty} \sqrt[n]{n} \\
a_n &= \sqrt[n]{n} - 1 \\
(a_n + 1) ^ n &= n \\
\binom{n}{2}a_n^{n-2} &\le n \\
a_n &\le \frac{2}{n-1}\\
\lim_{n \to \infty} \sqrt[n]{n} &= 1
\end{aligned}
$$
